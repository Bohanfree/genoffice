function restoreWpsBeautifyAlignmentJournal(journal, sheetId, marker, fileSha256) {
  // The native journal ignores ht/vt=null. Restore only this batch's previous
  // alignment deltas, and only while its original save baseline is current.
  if (!marker || marker.fileSha256 !== fileSha256) return;
  const entries = journal.cells.get(sheetId);
  if (!entries) return;
  for (const saved of marker.cells) {
    const key = `${saved.row}:${saved.column}`;
    const entry = entries.get(key);
    if (!entry) continue;
    const style = { ...entry.style };
    for (const field of ["horizontalAlignment", "verticalAlignment"]) {
      if (Object.prototype.hasOwnProperty.call(saved, field)) style[field] = saved[field];
      else delete style[field];
    }
    const next = { ...entry };
    if (Object.keys(style).length) next.style = style;
    else delete next.style;
    if (!next.hasValue && !next.style && !next.styleReset) entries.delete(key);
    else entries.set(key, next);
  }
}

  const targetNow = () => {
    const target = activeTarget$1(tableContext);
    if (!target) throw new Error("请先打开一个工作簿");
    if (origin && (target.state !== origin.state || target.sheetId !== origin.sheetId)) throw new Error("工作表已切换，请重新打开表格美化");

    const height = area.endRow - area.startRow + 1;
    const width = area.endColumn - area.startColumn + 1;
    if (area.startRow < 0 || area.startColumn < 0 || area.endRow >= 1048576 || area.endColumn >= 16384 || height < 1 || width < 1 || height * width > 100000) throw new Error("请选择 1 至 100000 个单元格的区域");
    return { area, height, width };
  };
  const preparedRange = async (target, text) => {
    const parsed = parseArea(text);
    await ensureRowsLoaded(tableContext, target, parsed.area);

    const nonempty = range.getCellDatas().some((row) => row.some((cell) => cell && (cell.v !== null && cell.v !== undefined && cell.v !== "" || cell.f || cell.si || cell.p?.body?.dataStream?.trim())));
    if (!nonempty) throw new Error("所选区域没有数据，请选择需要美化的表格");

      await preparedRange(origin, range);

    },
    apply: async ({ range: text, preset, options, buildStyleMatrix, measureLayout }) => {

          const cell = { row, column };
          for (const field of ["horizontalAlignment", "verticalAlignment"]) {
            if (style && Object.prototype.hasOwnProperty.call(style, field)) cell[field] = style[field];
          }
          alignmentBefore.cells.push(cell);
        }

      loadAutoHeightSuppression.active = true;

      const execute = (command, params) => {
        const success = target.runtime.univerAPI.syncExecuteCommand(command, { unitId, subUnitId: target.sheetId, ...params });
        const item = undoService._pitchUndoElement(unitId);
        if (item && item !== priorUndo && item.undoMutations.length > cursor) {
          undoGroups.push(item.undoMutations.slice(cursor));

        // Native batching appends groups; undo must replay commands in reverse order.
        if (item && item !== priorUndo) {
          item.undoMutations = undoGroups.reverse().flat();

      }

      ctx.setMessage(`已应用 ${preset.name} · ${height} 行 × ${width} 列，保存前可一次撤销`);
    }
  });
